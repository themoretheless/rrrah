//! Test-only spatial native composition in caller-verified blending coordinates.
//! Constant paints/groups and centroid-sampled fields; nonlinear integration,
//! masks, cross-space boundaries and full scene routing remain unqualified.
//! No display-color round trip.
use super::native_blend::Pixel;
use super::polygon_coverage::admitted_selected_polygon_pixel_row_budget;
use hayro_interpret::BlendMode;
use kurbo::{BezPath,Shape};
use vello_cpu::peniko::Fill;

#[derive(Clone,Copy)]
pub(super) struct SpatialPaint<'a> {
    /// Intersection of path and clip regions, indexed in the caller region list.
    pub(super) regions:&'a [usize],
    pub(super) source:Pixel,
    pub(super) mode:BlendMode,
}

/// Partition the whole pixel, compose ordered native paints per spatial cell,
/// then average premultiplied components. Refusal returns no partial result.
/// Polygon work and caller paint work have separate, explicit finite budgets.
pub(super) fn compose_polygon_pixel(
    regions:&[(&BezPath,Fill)],paints:&[SpatialPaint<'_>],backdrop:Pixel,
    x:i32,y:i32,polygon_work:usize,paint_work:usize,cancelled:&dyn Fn()->bool,
    admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<Pixel> {
    if cancelled() || regions.len()>4095 || paints.len()>4096 || !backdrop.valid()
        || x.abs_diff(0)>1_048_576 || y.abs_diff(0)>1_048_576 {return None;}
    for paint in paints {
        if cancelled() || paint.regions.is_empty() || paint.regions.len()>4096 || !paint.source.valid()
            || !matches!(paint.mode,BlendMode::Normal|BlendMode::Multiply|BlendMode::Screen|BlendMode::HardLight)
            || paint.source.model()!=backdrop.model()
            || paint.regions.iter().any(|&i|i>=regions.len()) {return None;}
    }
    let mut paint_remaining=paint_work;
    compose_cells(regions,backdrop,x,y,polygon_work,cancelled,admit,&mut |winding,_point,_coverage| {
        let mut pixel=backdrop;
        for paint in paints {
            paint_remaining=paint_remaining.checked_sub(paint.regions.len().checked_add(1)?)?;
            if cancelled() {return None;}
            let covered=paint.regions.iter().all(|&i|match regions[i].1 {
                Fill::NonZero=>winding[i]!=0,Fill::EvenOdd=>winding[i]%2!=0,
            });
            if covered {pixel=Pixel::composite(pixel,pixel,paint.source,1.,paint.mode)?;}
        }
        Some(pixel)
    })
}

fn compose_cells(
    regions:&[(&BezPath,Fill)],backdrop:Pixel,x:i32,y:i32,polygon_work:usize,
    cancelled:&dyn Fn()->bool,admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
    evaluate:&mut dyn FnMut(&[i32],kurbo::Point,&[u64;64])->Option<Pixel>,
)->Option<Pixel> {
    let bytes=(regions.len()+1).checked_mul(size_of::<(&BezPath,Fill)>()+size_of::<Option<usize>>())?
        .checked_add(5*size_of::<kurbo::PathEl>())?
        .checked_add(regions.len().checked_mul(size_of::<i32>()+size_of::<Option<usize>>())?)?;
    let _credit=admit(bytes)?;
    let unit=kurbo::Rect::new(f64::from(x),f64::from(y),f64::from(x)+1.,f64::from(y)+1.).to_path(0.1);
    let mut all=Vec::new();all.try_reserve_exact(regions.len()+1).ok()?;
    all.push((&unit,Fill::NonZero));
    let mut heads=Vec::<Option<usize>>::new();heads.try_reserve_exact(regions.len()+1).ok()?;heads.push(None);
    let mut next=Vec::<Option<usize>>::new();next.try_reserve_exact(regions.len()).ok()?;
    let mut remaining=polygon_work;
    // Exactly equal retained paths have identical winding, independently
    // of each alias's fill rule. Partition their geometry once; preserve all
    // original region IDs through an admitted linked alias list.
    for (index,&(path,fill)) in regions.iter().enumerate() {
        if cancelled() {return None;}
        let mut found=None;
        for (candidate,&(existing,_)) in all.iter().enumerate().skip(1) {
            remaining=remaining.checked_sub(1)?;
            let mut equal=std::ptr::eq(path,existing);
            if !equal && path.elements().len()==existing.elements().len() {
                equal=true;
                for (element,(a,b)) in path.elements().iter().zip(existing.elements()).enumerate() {
                    remaining=remaining.checked_sub(1)?;
                    if element%256==0 && cancelled() {return None;}
                    if a!=b {equal=false;break;}
                }
            }
            if equal {found=Some(candidate);break;}
        }
        let canonical=if let Some(found)=found {found} else {
            all.push((path,fill));heads.push(None);all.len()-1
        };
        next.push(heads[canonical]);heads[canonical]=Some(index);
    }
    if std::env::var_os("RRRAH_TRACE_SPATIAL_ROUTE").is_some() {
        eprintln!("SPATIAL_CANONICAL_PATHS original={} unique={}",regions.len(),all.len()-1);
    }
    let row=admitted_selected_polygon_pixel_row_budget(&all,x,y,&mut remaining,cancelled,admit).or_else(|| {
        if std::env::var_os("RRRAH_TRACE_SPATIAL_ROUTE").is_some() {eprintln!("SPATIAL_POLYGON_PREPARE_REFUSAL regions={} remaining={remaining}",all.len());}None
    })?;
    let mut saved=Vec::new();saved.try_reserve_exact(regions.len()).ok()?;
    saved.resize(regions.len(),0i32);
    let saved=std::cell::RefCell::new(saved);
    let coverage=std::cell::RefCell::new([0u64;64]);
    let mut alpha=0.;let mut integral=[0.;4];
    row.pixel_incremental_selected_fragments(&all,x,y,remaining,cancelled,admit,regions.len().max(1),&mut |winding,changed| {
        if cancelled() {return None;}
        let mut values=saved.borrow_mut();
        let mut bits=coverage.borrow_mut();
        let mut update=|canonical:usize| {
            let mut alias=heads[canonical];
            while let Some(i)=alias {
                values[i]=winding[canonical];
                let inside=match regions[i].1 {Fill::NonZero=>values[i]!=0,Fill::EvenOdd=>values[i]%2!=0};
                let mask=1u64<<(i%64);
                if inside {bits[i/64]|=mask;} else {bits[i/64]&=!mask;}
                alias=next[i];
            }
        };
        if let Some(region)=changed {update(region);} else {
            for canonical in 1..all.len() {update(canonical);}
        }
        Some(Some(0))
    },&mut |fragment| {
        let pixel=evaluate(&saved.borrow(),fragment.centroid,&coverage.borrow()).or_else(|| {
            if std::env::var_os("RRRAH_TRACE_SPATIAL_ROUTE").is_some() {eprintln!("SPATIAL_CELL_EVALUATION_REFUSAL point={:?}",fragment.centroid);}None
        })?;
        if pixel.model()!=backdrop.model() || !pixel.valid() {return None;}
        let weight=fragment.area*pixel.alpha();alpha+=weight;
        for (sum,value) in integral.iter_mut().zip(pixel.native_components()?) {*sum+=weight*value;}
        Some(())
    }).or_else(|| {
        if std::env::var_os("RRRAH_TRACE_SPATIAL_ROUTE").is_some() {eprintln!("SPATIAL_POLYGON_FRAGMENTS_REFUSAL regions={}",all.len());}None
    })?;
    if !alpha.is_finite() || !(0. ..=1.+1e-12).contains(&alpha) {return None;}
    let components=if alpha==0. {[0.;4]} else {integral.map(|v|v/alpha)};
    Pixel::from_native(backdrop.model(),&components[..backdrop.model().channels()],alpha.min(1.))
}

pub(super) type SpatialSampler<'a>=dyn Fn(kurbo::Point,&dyn Fn()->bool)->Option<Option<Pixel>>+'a;

pub(super) type SpatialMaskSampler<'a>=dyn Fn(kurbo::Point,&dyn Fn()->bool)->Option<f64>+'a;

pub(super) struct NativeSpatialSample<'a> {
    pub(super) pixel:Pixel,
    pub(super) space:&'a hayro_interpret::color::ColorSpace,
}
pub(super) type NativeSpatialSampler<'a>=dyn Fn(kurbo::Point,&dyn Fn()->bool,&mut usize,Option<usize>)->Option<Option<NativeSpatialSample<'a>>>+'a;

/// A mask shares the caller's remaining work; it must debit before doing work.
pub(super) type BudgetedMaskSampler<'a>=dyn Fn(kurbo::Point,&dyn Fn()->bool,&mut usize)->Option<f64>+'a;

#[derive(Clone,Copy)]
pub(super) enum SpatialCommand<'a> {
    RetainedPaint {regions:&'a [usize],paint:&'a super::native_paint::NativePaint,mode:BlendMode},
    // Owners must be ordered by descending original triangle index.
    NativeSampledPaint {regions:&'a [usize],sample:&'a NativeSpatialSampler<'a>,owners:&'a [(usize,usize)],alpha_is_shape:bool,mode:BlendMode},
    BeginInSpace {space:&'a hayro_interpret::color::ColorSpace,knockout:bool,opacity:f64,alpha_is_shape:bool,mode:BlendMode},
    /// Mask sampler supplies native alpha or separately verified luminosity;
    /// all mask boundaries must be included in the spatial arrangement.
    SetMask {sample:&'a SpatialMaskSampler<'a>,sample_cost:usize},
    SetBudgetedMask {sample:&'a BudgetedMaskSampler<'a>},
    ClearMask,
    Paint(SpatialPaint<'a>),
    /// Caller must include every source boundary in the region arrangement.
    /// Centroid integration is exact only for affine final premultiplied fields;
    /// nonlinear functions/blending need further integration, not qualified here.
    SampledPaint {regions:&'a [usize],sample:&'a SpatialSampler<'a>,sample_cost:usize,alpha_is_shape:bool,mode:BlendMode},
    Begin {isolated:bool,knockout:bool,opacity:f64,alpha_is_shape:bool,mode:BlendMode},
    End,
}

/// Remove groups with no paint commands, preserving all state outside them.
/// Uses the existing admitted command storage and a bounded stack.
pub(super) fn prune_empty_groups(commands:&mut Vec<SpatialCommand<'_>>,cancelled:&dyn Fn()->bool)->Option<()> {
    let mut stack=[(0usize,false);32];let mut depth=0usize;let mut out=0usize;
    for index in 0..commands.len() {
        if cancelled() {return None;}
        let command=commands[index];
        match command {
            SpatialCommand::Begin {..}|SpatialCommand::BeginInSpace {..}=>{
                if depth==stack.len() {return None;}
                stack[depth]=(out,false);depth+=1;
            },
            SpatialCommand::End=>{
                depth=depth.checked_sub(1)?;
                let (start,painted)=stack[depth];
                if !painted {out=start;continue;}
                if depth>0 {stack[depth-1].1=true;}
            },
            SpatialCommand::Paint(_)|SpatialCommand::RetainedPaint {..}|
            SpatialCommand::SampledPaint {..}|SpatialCommand::NativeSampledPaint {..}=>{
                if depth>0 {stack[depth-1].1=true;}
            },
            _=>{},
        }
        commands[out]=command;out+=1;
    }
    if depth!=0 {return None;}
    commands.truncate(out);Some(())
}

/// Native group lifecycle is evaluated independently in each spatial cell.
/// Sources and group space must already share verified blending coordinates.
/// Native mask source/transfer construction and cross-space boundaries require
/// their separate scene router. Begin captures the parent mask for group exit,
/// resets the child mask, and End restores the parent mask.
pub(super) fn compose_grouped_polygon_pixel(
    regions:&[(&BezPath,Fill)],commands:&[SpatialCommand<'_>],
    space:&hayro_interpret::color::ColorSpace,backdrop:Pixel,x:i32,y:i32,
    polygon_work:usize,command_work:usize,cancelled:&dyn Fn()->bool,
    admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<Pixel> {
    compose_grouped_polygon_impl(regions,commands,space,backdrop,x,y,None,polygon_work,&mut {command_work},cancelled,admit)
}
/// Evaluate native source/group state at one point without area averaging.
/// Mask callers must include the mask's geometry in the parent partition.
pub(super) fn compose_grouped_polygon_point(
    regions:&[(&BezPath,Fill)],commands:&[SpatialCommand<'_>],
    space:&hayro_interpret::color::ColorSpace,backdrop:Pixel,point:kurbo::Point,
    geometry_work:usize,command_work:usize,cancelled:&dyn Fn()->bool,
    admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<Pixel> {
    compose_grouped_polygon_impl(regions,commands,space,backdrop,0,0,Some(point),geometry_work,&mut {command_work},cancelled,admit)
}

pub(super) fn compose_grouped_polygon_impl(
    regions:&[(&BezPath,Fill)],commands:&[SpatialCommand<'_>],
    space:&hayro_interpret::color::ColorSpace,backdrop:Pixel,x:i32,y:i32,point_override:Option<kurbo::Point>,
    polygon_work:usize,remaining:&mut usize,cancelled:&dyn Fn()->bool,
    admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<Pixel> {
    use super::native_frame::PaintFrame;
    struct WorkTrace {enabled:bool,counts:[std::cell::Cell<usize>;13],dispatch:[std::cell::Cell<usize>;8],spent:[std::cell::Cell<usize>;5]}
    impl WorkTrace {
        fn measure<T>(&self,index:usize,remaining:&mut usize,run:impl FnOnce(&mut usize)->Option<T>)->Option<T> {
            if !self.enabled {return run(remaining);}
            let before=*remaining;let result=run(remaining);
            let used=before.checked_sub(*remaining).unwrap_or(0);
            self.spent[index].set(self.spent[index].get()+used);result
        }
        fn kind(command:&SpatialCommand<'_>)->usize {
            match command {
                SpatialCommand::RetainedPaint {..}=>0,SpatialCommand::Paint(_)=>1,
                SpatialCommand::NativeSampledPaint {..}=>2,SpatialCommand::SampledPaint {..}=>3,
                SpatialCommand::Begin {..}|SpatialCommand::BeginInSpace {..}=>4,
                SpatialCommand::End=>5,SpatialCommand::ClearMask=>6,_=>7,
            }
        }
        fn command(&self,command:&SpatialCommand<'_>) {
            if self.enabled {let index=Self::kind(command);self.dispatch[index].set(self.dispatch[index].get()+1);}
        }
        fn hit(&self,index:usize) {if self.enabled {self.counts[index].set(self.counts[index].get()+1);}}
    }
    impl Drop for WorkTrace {
        fn drop(&mut self) {
            if self.enabled {
                let b=self.spent.each_ref().map(|v|v.get());
                eprintln!("SPATIAL_BUDGET_SPENT key={} lookup={} route_preparation={} native_source={} backdrop_comparison={}",b[0],b[1],b[2],b[3],b[4]);
                let c=self.counts.each_ref().map(|v|v.get());
                let d=self.dispatch.each_ref().map(|v|v.get());
                eprintln!("SPATIAL_DISPATCH_KINDS retained={} paint={} native={} sampled={} begin={} end={} clear_mask={} set_mask={}",d[0],d[1],d[2],d[3],d[4],d[5],d[6],d[7]);
                eprintln!("SPATIAL_WORK_COUNTS fragments={} route_hits={} route_misses={} dispatch={} groups={} native_samples={} fixed_samples={} mask_samples={} constant_hits={} group_hits={} paint_run_hits={} hidden_commands={} local_signature_hits={}",c[0],c[1],c[2],c[3],c[4],c[5],c[6],c[7],c[8],c[9],c[10],c[11],c[12]);
            }
        }
    }
    let work_trace=WorkTrace {enabled:std::env::var_os("RRRAH_TRACE_SPATIAL_ROUTE").is_some(),
        counts:std::array::from_fn(|_|std::cell::Cell::new(0)),dispatch:std::array::from_fn(|_|std::cell::Cell::new(0)),spent:std::array::from_fn(|_|std::cell::Cell::new(0))};
    if work_trace.enabled {
        let mut kinds=[0usize;8];
        for (index,command) in commands.iter().enumerate() {
            kinds[WorkTrace::kind(command)]+=1;
            match command {
                SpatialCommand::Begin {isolated,knockout,opacity,alpha_is_shape,mode}=>
                    eprintln!("SPATIAL_GROUP_PROPERTIES command={index} isolated={isolated} knockout={knockout} opacity={opacity} ais={alpha_is_shape} mode={mode:?}"),
                SpatialCommand::BeginInSpace {knockout,opacity,alpha_is_shape,mode,..}=>
                    eprintln!("SPATIAL_GROUP_PROPERTIES command={index} isolated=true explicit_space=true knockout={knockout} opacity={opacity} ais={alpha_is_shape} mode={mode:?}"),_=>{},
            }
        }
        eprintln!("SPATIAL_PREPARED_KINDS retained={} paint={} native={} sampled={} begin={} end={} clear_mask={} set_mask={}",kinds[0],kinds[1],kinds[2],kinds[3],kinds[4],kinds[5],kinds[6],kinds[7]);
    }
    if cancelled() || regions.len()>4095 || commands.len()>4096 || !backdrop.valid()
        || x.abs_diff(0)>1_048_576 || y.abs_diff(0)>1_048_576 {return None;}
    let valid_mode=|mode| {
        let valid=matches!(mode,BlendMode::Normal|BlendMode::Multiply|BlendMode::Screen|BlendMode::HardLight);
        if !valid && std::env::var_os("RRRAH_TRACE_SPATIAL_ROUTE").is_some() {eprintln!("SPATIAL_UNSUPPORTED_BLEND mode={mode:?}");}
        valid
    };
    let mut depth=0usize;let mut maximum=0;
    for command in commands {
        if cancelled() {return None;}
        match command {
            SpatialCommand::SetMask {sample_cost,..}=>{
                if !(1..=1_000_000).contains(sample_cost) {return None;}
            },
            SpatialCommand::ClearMask|SpatialCommand::SetBudgetedMask {..}=>{},
            SpatialCommand::Begin {opacity,mode,..}|SpatialCommand::BeginInSpace {opacity,mode,..}=>{
                if !opacity.is_finite() || !(0. ..=1.).contains(opacity) || !valid_mode(*mode) {return None;}
                depth+=1;if depth>32 {return None;}maximum=maximum.max(depth);
            },
            SpatialCommand::End=>{depth=depth.checked_sub(1)?;},
            SpatialCommand::SampledPaint {regions:required,sample_cost,mode,..}=>{
                if required.is_empty() || required.len()>4096 || !valid_mode(*mode)
                    || !(1..=1_000_000).contains(sample_cost)
                    || required.iter().any(|&i|i>=regions.len()) {return None;}
            },
            SpatialCommand::RetainedPaint {regions:required,mode,..}|
            SpatialCommand::NativeSampledPaint {regions:required,mode,..}=>{
                if required.is_empty() || required.len()>4096 || !valid_mode(*mode)
                    || required.iter().any(|&i|i>=regions.len()) {return None;}
            },
            SpatialCommand::Paint(paint)=>{
                if paint.regions.is_empty() || paint.regions.len()>4096 || !paint.source.valid()
                    || paint.source.model()!=backdrop.model() || !valid_mode(paint.mode)
                    || paint.regions.iter().any(|&i|i>=regions.len()) {return None;}
            },
        }
    }
    for command in commands {
        if let SpatialCommand::NativeSampledPaint {owners,..}=command {
            let mut previous=None;
            for &(region,index) in *owners {
                if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
                if region>=regions.len() || previous.is_some_and(|old|old<index) {return None;}
                previous=Some(index);
            }
        }
    }
    if depth!=0 {return None;}
    let words=regions.len().checked_add(63)?/64;
    let _ends_credit=admit(commands.len().checked_mul(size_of::<usize>())?)?;
    let mut ends=Vec::new();ends.try_reserve_exact(commands.len()).ok()?;ends.resize(commands.len(),0usize);
    let mut routing_bits=[0u64;64];
    let mut ancestors=[0usize;32];let mut ancestor_depth=0usize;
    for (index,command) in commands.iter().enumerate() {
        if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
        match command {
            SpatialCommand::Begin {..}|SpatialCommand::BeginInSpace {..}=>{ancestors[ancestor_depth]=index;ancestor_depth+=1;},
            SpatialCommand::End=>{ancestor_depth=ancestor_depth.checked_sub(1)?;ends[ancestors[ancestor_depth]]=index;},
            SpatialCommand::Paint(_)|SpatialCommand::RetainedPaint {..}|SpatialCommand::SampledPaint {..}|SpatialCommand::NativeSampledPaint {..}=>{
                let required=match command {SpatialCommand::Paint(p)=>p.regions,
                    SpatialCommand::RetainedPaint {regions,..}|SpatialCommand::SampledPaint {regions,..}|
                    SpatialCommand::NativeSampledPaint {regions,..}=>*regions,_=>return None};
                for &region in required {*remaining=remaining.checked_sub(1)?;routing_bits[region/64]|=1u64<<(region%64);}
            },_=>{},
        }
    }
    let mut constant_key=[0u64;64];let mut constant_regions=0usize;
    if point_override.is_none() {
        for word in 0..words {
            let mut needed=routing_bits[word];
            while needed!=0 {
                if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
                let bit=needed.trailing_zeros() as usize;needed&=needed-1;
                if rectangle_strictly_contains_pixel(regions[word*64+bit].0,x,y) {
                    routing_bits[word]&=!(1u64<<bit);constant_key[word]|=1u64<<bit;constant_regions+=1;
                }
            }
        }
    }
    if work_trace.enabled {eprintln!("SPATIAL_CONSTANT_RECTANGLE_REGIONS={constant_regions}");}
    if std::env::var_os("RRRAH_TRACE_SPATIAL_ROUTE").is_some() {
        eprintln!("SPATIAL_ROUTING_REGIONS total={} used={}",regions.len(),routing_bits.iter().map(|v|v.count_ones() as usize).sum::<usize>());
    }
    type State=super::native_frame::SinglePixelState;
    type Credit=Box<dyn std::any::Any+Send+Sync>;
    struct Transition {input:State,output:State,_credit:Credit}
    struct GroupSlot {end:Option<usize>,frame:Option<Box<PaintFrame>>,
        run_end:Option<usize>,transition:Option<Box<Transition>>}
    struct Route {key:[u64;64],commands:Vec<usize>,groups:Vec<GroupSlot>,constant:bool,result:Option<Pixel>,hash:u64,reuse_bits:Vec<u64>}
    const ROUTE_INDEX_SIZE:usize=128;
    type Index=[Option<(u64,usize)>;ROUTE_INDEX_SIZE];
    fn insert_index(table:&mut Index,hash:u64,index:usize,remaining:&mut usize,cancelled:&dyn Fn()->bool)->Option<()> {
        let mut slot=hash as usize%ROUTE_INDEX_SIZE;
        for _ in 0..ROUTE_INDEX_SIZE {
            if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
            if table[slot].is_none() {table[slot]=Some((hash,index));return Some(());}
            slot=(slot+1)%ROUTE_INDEX_SIZE;
        }
        None
    }
    let _index_credit=admit(size_of::<Index>())?;let mut route_lookup:Index=[None;ROUTE_INDEX_SIZE];
    const ROUTE_LIMIT:usize=64;
    let route_bytes=ROUTE_LIMIT.checked_mul(size_of::<Route>().checked_add(commands.len().checked_mul(size_of::<usize>().checked_add(size_of::<GroupSlot>())?)?)?.checked_add(words.checked_mul(size_of::<u64>())?)?)?
        .checked_add(commands.len().checked_mul(2)?)?;
    let _route_credit=admit(route_bytes)?;
    let mut routes=Vec::<Route>::new();routes.try_reserve_exact(ROUTE_LIMIT).ok()?;
    let mut keep=Vec::new();keep.try_reserve_exact(commands.len()).ok()?;keep.resize(commands.len(),true);
    let mut ignored=Vec::new();ignored.try_reserve_exact(commands.len()).ok()?;ignored.resize(commands.len(),false);
    type Entry=(Box<PaintFrame>,Option<(f64,bool,BlendMode,Option<usize>,usize)>);
    let _mask_cache_credit=admit(commands.len().checked_mul(size_of::<Option<f64>>())?)?;
    let mut mask_cache=Vec::new();mask_cache.try_reserve_exact(commands.len()).ok()?;
    mask_cache.resize(commands.len(),None);
    // Each command has a fixed enclosing group-space lifecycle. Keep its
    // constant native conversion across fragments; apply_prepared still verifies
    // the destination space on every use. Sampled sources are never cached here.
    let _paint_cache_credit=admit(commands.len().checked_mul(size_of::<Option<super::native_frame::PreparedPaint>>())?)?;
    let mut paint_cache=Vec::new();paint_cache.try_reserve_exact(commands.len()).ok()?;
    paint_cache.resize_with(commands.len(),||None);
    // Point mode builds the coverage bits once below. Cell mode charges each
    // incremental update through the polygon selector's declared work cost.
    if point_override.is_some() {*remaining=remaining.checked_sub(regions.len())?;}
    let mut last_route:Option<usize>=None;
    let mut evaluate=|winding:&[i32],point:kurbo::Point,coverage:&[u64;64]| {

    use super::native_blend::Model;

#[test]
fn spatial_clips_native_cmyk_and_refusals_keep_admission_bounded() {
    use super::native_blend::Model;
    use std::sync::{Arc,atomic::{AtomicUsize,Ordering}};
    struct Credit(Arc<AtomicUsize>,usize);
    impl Drop for Credit {fn drop(&mut self) {self.0.fetch_sub(self.1,Ordering::Relaxed);}}
    let used=Arc::new(AtomicUsize::new(0));
    let requests=std::cell::Cell::new(0);let deny=std::cell::Cell::new(usize::MAX);
    let admit=|bytes| {requests.set(requests.get()+1);if requests.get()==deny.get() {return None;}
        used.fetch_add(bytes,Ordering::Relaxed);Some(Box::new(Credit(used.clone(),bytes)) as Box<dyn std::any::Any+Send+Sync>)};
    let path=kurbo::Rect::new(0.,0.,1.,1.).to_path(0.1);
    let clip=kurbo::Rect::new(0.1,0.,0.6,1.).to_path(0.1);
    let regions=[(&path,Fill::NonZero),(&clip,Fill::EvenOdd)];
    let source=Pixel::from_native(Model::Cmyk,&[0.25,0.5,0.75,0.125],0.5).unwrap();
    let clear=Pixel::from_native(Model::Cmyk,&[0.;4],0.).unwrap();
    let paints=[SpatialPaint {regions:&[0,1],source,mode:BlendMode::Normal}];
    let run=|polygon,paint,cancelled:&dyn Fn()->bool|compose_polygon_pixel(&regions,&paints,clear,
        0,0,polygon,paint,cancelled,&admit);
    let polls=std::cell::Cell::new(0);
    let pixel=run(1_000_000,1_000_000,&||{polls.set(polls.get()+1);false}).unwrap();
    assert!((pixel.alpha()-0.25).abs()<1e-12);
    assert_eq!(pixel.native_components().unwrap(),[0.25,0.5,0.75,0.125]);
    assert_eq!(used.load(Ordering::Relaxed),0);
    let total=requests.get();
    for failure in 1..=total {
        requests.set(0);deny.set(failure);assert!(run(1_000_000,1_000_000,&||false).is_none());
        assert_eq!(used.load(Ordering::Relaxed),0);
    }
    deny.set(usize::MAX);
    for stop in 1..=polls.get() {
        let current=std::cell::Cell::new(0);
        assert!(run(1_000_000,1_000_000,&||{current.set(current.get()+1);current.get()==stop}).is_none());
        assert_eq!(used.load(Ordering::Relaxed),0);
    }
    assert!(run(2,1_000_000,&||false).is_none());
    assert!(run(1_000_000,0,&||false).is_none());
    assert_eq!(used.load(Ordering::Relaxed),0);
}

pub(super) struct RasterCells {
    /// First path is the image footprint; remaining paths partition texels.
    pub(super) paths:Vec<BezPath>,
    _credit:Box<dyn std::any::Any+Send+Sync>,
}
/// Nearest-neighbor raster geometry, conservatively selected by inverse bounds.
/// Original component storage stays with the source. Bilinear/nonlinear source
/// integration is separate; callers must not use this as its qualification.
pub(super) fn admitted_raster_pixel_cells(
    width:u32,height:u32,transform:kurbo::Affine,rect:kurbo::Rect,work_limit:usize,
    cancelled:&dyn Fn()->bool,admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
 )->Option<RasterCells> {
    admitted_raster_pixel_cells_budget(width,height,transform,rect,&mut {work_limit},cancelled,admit)
}
fn admitted_raster_pixel_cells_budget(
    width:u32,height:u32,transform:kurbo::Affine,rect:kurbo::Rect,remaining:&mut usize,
    cancelled:&dyn Fn()->bool,admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<RasterCells> {
    if cancelled() || width==0 || height==0
        || [rect.x0,rect.y0,rect.x1,rect.y1].iter().any(|v|!v.is_finite())
        || rect.width()<=0. || rect.height()<=0. {return None;}
    let determinant=transform.determinant();
    if !determinant.is_finite() || determinant==0. {return None;}
    let query=transform.inverse().transform_rect_bbox(rect);
    if [query.x0,query.y0,query.x1,query.y1].iter().any(|v|!v.is_finite()) {return None;}
    let x0=query.x0.floor().clamp(0.,f64::from(width)) as u32;
    let x1=query.x1.ceil().clamp(0.,f64::from(width)) as u32;
    let y0=query.y0.floor().clamp(0.,f64::from(height)) as u32;
    let y1=query.y1.ceil().clamp(0.,f64::from(height)) as u32;
    let count=usize::try_from(x1.checked_sub(x0)?).ok()?.checked_mul(usize::try_from(y1.checked_sub(y0)?).ok()?)?.checked_add(1)?;
    if count>4095 {return None;}
    let bytes=count.checked_mul(size_of::<BezPath>().checked_add(5*size_of::<kurbo::PathEl>())?)?;
    let credit=admit(bytes)?;
    let mut paths=Vec::new();paths.try_reserve_exact(count).ok()?;
    let mut emit=|bounds:kurbo::Rect|->Option<()> {
        if cancelled() {return None;}
        *remaining=remaining.checked_sub(1)?;
        let mut elements=Vec::new();elements.try_reserve_exact(5).ok()?;
        for point in [kurbo::Point::new(bounds.x0,bounds.y0),kurbo::Point::new(bounds.x1,bounds.y0),
            kurbo::Point::new(bounds.x1,bounds.y1),kurbo::Point::new(bounds.x0,bounds.y1)] {
            *remaining=remaining.checked_sub(1)?;if cancelled() {return None;}
            let point=transform*point;
            if !point.x.is_finite() || !point.y.is_finite() || point.x.abs()>1_048_576. || point.y.abs()>1_048_576. {return None;}
            elements.push(if elements.is_empty() {kurbo::PathEl::MoveTo(point)} else {kurbo::PathEl::LineTo(point)});
        }
        elements.push(kurbo::PathEl::ClosePath);paths.push(BezPath::from_vec(elements));Some(())
    };
    emit(kurbo::Rect::new(0.,0.,f64::from(width),f64::from(height)))?;
    for y in y0..y1 {for x in x0..x1 {emit(kurbo::Rect::new(f64::from(x),f64::from(y),f64::from(x)+1.,f64::from(y)+1.))?;}}
    Some(RasterCells {paths,_credit:credit})
}

/// Retain both source grids under one aggregate geometry-work allowance.
/// The mask maps to the same footprint even when its resolution differs.
pub(super) fn admitted_raster_mask_cells(
    width:u32,height:u32,mask_width:u32,mask_height:u32,transform:kurbo::Affine,
    rect:kurbo::Rect,work_limit:usize,cancelled:&dyn Fn()->bool,
    admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<(RasterCells,RasterCells)> {
    if width==0 || height==0 || mask_width==0 || mask_height==0 {return None;}
    let mut work=work_limit;
    let color=admitted_raster_pixel_cells_budget(width,height,transform,rect,&mut work,cancelled,admit)?;
    let mask_transform=transform*kurbo::Affine::scale_non_uniform(
        f64::from(width)/f64::from(mask_width),f64::from(height)/f64::from(mask_height));
    let mask=admitted_raster_pixel_cells_budget(mask_width,mask_height,mask_transform,rect,&mut work,cancelled,admit)?;
    if color.paths.len().checked_add(mask.paths.len())?>4095 {return None;}
    Some((color,mask))
}

impl RasterCells {
    pub(super) fn into_parts(self)->(Vec<BezPath>,Box<dyn std::any::Any+Send+Sync>) {
        (self.paths,self._credit)
    }
}
/// Partition a finite pixel rectangle by infinite source boundary lines.
/// Each path is one clipped half-plane; there is no implied source footprint.
pub(super) fn admitted_line_partition_cells(
    lines:&[[kurbo::Point;2]],rect:kurbo::Rect,work_limit:usize,cancelled:&dyn Fn()->bool,
    admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<RasterCells> {
    if cancelled() || lines.len()>64 || rect.width()<=0. || rect.height()<=0.
        || [rect.x0,rect.x1,rect.y0,rect.y1].iter().any(|v|!v.is_finite() || v.abs()>1_048_576.) {return None;}
    let credit=admit(lines.len().checked_mul(size_of::<BezPath>()+6*size_of::<kurbo::PathEl>())?)?;
    let mut paths=Vec::new();paths.try_reserve_exact(lines.len()).ok()?;
    let mut work=work_limit;
    for [a,b] in lines {
        if cancelled() {return None;}
        work=work.checked_sub(20)?;
        let d=*b-*a;
        if [a.x,a.y,b.x,b.y,d.x,d.y].iter().any(|v|!v.is_finite()) || d.hypot2()==0. {return None;}
        let corners=[kurbo::Point::new(rect.x0,rect.y0),kurbo::Point::new(rect.x1,rect.y0),
            kurbo::Point::new(rect.x1,rect.y1),kurbo::Point::new(rect.x0,rect.y1)];
        let side=|p:kurbo::Point|{let q=p-*a;d.x*q.y-d.y*q.x};
        let mut points=[kurbo::Point::ZERO;8];let mut count=0;
        for index in 0..4 {
            if cancelled() {return None;}
            let p=corners[index];let q=corners[(index+1)%4];let sp=side(p);let sq=side(q);
            if !sp.is_finite() || !sq.is_finite() {return None;}
            if sp>=0. {points[count]=p;count+=1;}
            if (sp<0. && sq>0.) || (sp>0. && sq<0.) {
                let t=sp/(sp-sq);let crossing=p+(q-p)*t;
                if !t.is_finite() || !crossing.x.is_finite() || !crossing.y.is_finite() {return None;}
                points[count]=crossing;count+=1;
            }
        }
        let mut elements=Vec::new();elements.try_reserve_exact(6).ok()?;
        if count>=3 {
            for (index,p) in points[..count].iter().enumerate() {elements.push(if index==0 {kurbo::PathEl::MoveTo(*p)} else {kurbo::PathEl::LineTo(*p)});}
            elements.push(kurbo::PathEl::ClosePath);
        }
        paths.push(BezPath::from_vec(elements));
    }
    Some(RasterCells {paths,_credit:credit})
}

/// Canonicalize only structurally equal paths with the same fill rule.
/// Hash collisions are checked by full equality and never merge distinct paths.
pub(super) fn deduplicate_regions(
    regions:&mut Vec<(&BezPath,Fill)>,requirements:&mut [Vec<usize>],
    cancelled:&dyn Fn()->bool,admit:&dyn Fn(usize)->Option<Box<dyn std::any::Any+Send+Sync>>,
)->Option<()> {
    use std::hash::{Hash,Hasher};
    let count=regions.len();if count>4095 {return None;}
    let _credit=admit(count.checked_mul(size_of::<(u64,usize)>()+2*size_of::<usize>())?)?;
    let mut order=Vec::<(u64,usize)>::new();order.try_reserve_exact(count).ok()?;
    let mut map=Vec::new();map.try_reserve_exact(count).ok()?;map.resize(count,0);
    let mut representatives=Vec::new();representatives.try_reserve_exact(count).ok()?;
    let mut work=4_000_000usize;
    for (index,(path,fill)) in regions.iter().enumerate() {
        if let Some(&(hash,_))=order.iter().find(|&&(_,other)|
            std::ptr::eq(regions[other].0,*path) && regions[other].1==*fill) {
            order.push((hash,index));continue;
        }
        let mut hash=std::collections::hash_map::DefaultHasher::new();
        matches!(fill,Fill::EvenOdd).hash(&mut hash);
        for element in path.elements() {
            if cancelled() {return None;}let Some(next)=work.checked_sub(1) else {return Some(());};work=next;
            let mut point=|p:kurbo::Point|{p.x.to_bits().hash(&mut hash);p.y.to_bits().hash(&mut hash);};
            match element {
                kurbo::PathEl::MoveTo(p)=>{point(*p);0u8.hash(&mut hash);},
                kurbo::PathEl::LineTo(p)=>{point(*p);1u8.hash(&mut hash);},
                kurbo::PathEl::QuadTo(a,b)=>{point(*a);point(*b);2u8.hash(&mut hash);},
                kurbo::PathEl::CurveTo(a,b,c)=>{point(*a);point(*b);point(*c);3u8.hash(&mut hash);},
                kurbo::PathEl::ClosePath=>4u8.hash(&mut hash),
            }
        }
        order.push((hash.finish(),index));
    }
    order.sort_unstable();let mut bucket_start=0;
    for position in 0..order.len() {
        if cancelled() {return None;}
        if position==0 || order[position].0!=order[position-1].0 {bucket_start=position;}
        let index=order[position].1;let mut existing=None;
        for &(_,other) in &order[bucket_start..position] {
            if !std::ptr::eq(regions[index].0,regions[other].0) {
                let Some(next)=work.checked_sub(regions[index].0.elements().len().checked_add(1)?) else {return Some(());};work=next;
            }
            if regions[index].1==regions[other].1 && regions[index].0.elements()==regions[other].0.elements() {
                existing=Some(map[other]);break;
            }
        }
        map[index]=if let Some(existing)=existing {existing} else {
            let target=representatives.len();representatives.push(index);target
        };
    }
    for required in requirements {
        for index in required.iter_mut() {if cancelled() {return None;}*index=*map.get(*index)?;}
        required.sort_unstable();required.dedup();
    }
    // Copy representatives before compacting, since canonical order is by hash.
    let _copy_credit=admit(representatives.len().checked_mul(size_of::<(&BezPath,Fill)>())?)?;
    let mut unique=Vec::new();unique.try_reserve_exact(representatives.len()).ok()?;
    for index in representatives {unique.push(regions[index]);}
    regions.clear();regions.extend(unique);Some(())
}

    let a=kurbo::Rect::new(0.,0.,0.25,1.).to_path(0.1);let copy=a.clone();

/// Consecutive opaque normal paints are idempotent only without a soft mask
/// and when original components already share the active blending coordinates.
pub(super) fn coalesce_opaque_paints(commands:&mut Vec<SpatialCommand<'_>>,
    space:&hayro_interpret::color::ColorSpace,cancelled:&dyn Fn()->bool)->Option<()> {
    let mut spaces=[space;33];let mut masks=[false;33];let mut depth=0usize;let mut out=0;
    let mut previous:Option<SpatialCommand<'_>>=None;
    for index in 0..commands.len() {
        if cancelled() {return None;}
        let command=commands[index];
        let duplicate=match (previous,command) {
            (Some(SpatialCommand::RetainedPaint {regions:a,paint:p,mode:BlendMode::Normal}),
                SpatialCommand::RetainedPaint {regions:b,paint:q,mode:BlendMode::Normal})=>{
                !masks[depth] && p.opacity==1. && q.opacity==1. && a==b && p.components==q.components
                    && p.color_space.shares_blending_coordinates(spaces[depth])
                    && q.color_space.shares_blending_coordinates(spaces[depth])
            },_=>false,
        };
        if duplicate {continue;}
        match command {
            SpatialCommand::Begin {..}=>{
                if depth==32 {return None;}spaces[depth+1]=spaces[depth];depth+=1;masks[depth]=false;
            },
            SpatialCommand::BeginInSpace {space,..}=>{
                if depth==32 {return None;}depth+=1;spaces[depth]=space;masks[depth]=false;
            },
            SpatialCommand::End=>depth=depth.checked_sub(1)?,
            SpatialCommand::SetMask {..}|SpatialCommand::SetBudgetedMask {..}=>masks[depth]=true,
            SpatialCommand::ClearMask=>masks[depth]=false,
            _=>{},
        }
        commands[out]=command;out+=1;previous=Some(command);
    }
    if depth!=0 {return None;}commands.truncate(out);Some(())
}

/// Flatten only inherited non-knockout neutral groups in a mask-free command
/// stream and non-knockout parent. All other object boundaries remain intact.
pub(super) fn flatten_neutral_groups(
    commands:&mut Vec<SpatialCommand<'_>>,remaining:&mut usize,cancelled:&dyn Fn()->bool,
)->Option<usize> {
    for command in commands.iter() {
        if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
        if matches!(command,SpatialCommand::SetMask {..}|SpatialCommand::SetBudgetedMask {..}) {return Some(0);}
    }
    let mut stack=[(false,false);33];let mut depth=0usize;let mut out=0usize;let mut removed=0usize;
    for index in 0..commands.len() {
        if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
        let command=commands[index];
        let skip=match command {
            SpatialCommand::Begin {isolated,knockout,opacity,alpha_is_shape,mode}=>{
                let flatten=!stack[depth].0 && !isolated && !knockout && opacity==1.
                    && !alpha_is_shape && mode==BlendMode::Normal;
                depth+=1;if depth>=stack.len() {return None;}stack[depth]=(knockout,flatten);flatten
            },
            SpatialCommand::BeginInSpace {knockout,..}=>{
                depth+=1;if depth>=stack.len() {return None;}stack[depth]=(knockout,false);false
            },
            SpatialCommand::End=>{
                if depth==0 {return None;}let skip=stack[depth].1;depth-=1;skip
            },_=>false,
        };
        if skip {removed+=1;} else {commands[out]=command;out+=1;}
    }
    if depth!=0 {return None;}commands.truncate(out);Some(removed)
}

/// Drop only objects hidden by a certified opaque Normal overwrite at this
/// exact coverage route. A child's certificate also requires neutral exit.
pub(super) fn prune_overwritten_route(
    commands:&[SpatialCommand<'_>],keep:&mut [bool],ignored:&mut [bool],ends:&[usize],
    remaining:&mut usize,cancelled:&dyn Fn()->bool,
)->Option<usize> {
    if commands.len()!=keep.len() || commands.len()!=ends.len() || commands.len()!=ignored.len() {return None;}
    ignored.fill(false);
    for command in commands {
        if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
        if matches!(command,SpatialCommand::SetMask {..}|SpatialCommand::SetBudgetedMask {..}) {return Some(0);}
    }
    // Earliest remaining object, group begin, knockout, independent opaque state.
    let mut frames=[(0usize,0usize,false,false);33];let mut depth=0usize;
    let mut index=0usize;let mut removed=0usize;
    while index<commands.len() {
        if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
        if !keep[index] {
            if matches!(commands[index],SpatialCommand::Begin {..}|SpatialCommand::BeginInSpace {..}) {
                index=ends[index].checked_add(1)?;continue;
            }
            index+=1;continue;
        }
        let mut overwrite=None;
        match commands[index] {
            SpatialCommand::Begin {knockout,..}|SpatialCommand::BeginInSpace {knockout,..}=>{
                depth+=1;if depth>=frames.len() {return None;}
                frames[depth]=(index+1,index,knockout,false);
            },
            SpatialCommand::End=>{
                if depth==0 {return None;}
                let (_,begin,_,independent)=frames[depth];depth-=1;
                let neutral=match commands[begin] {
                    SpatialCommand::Begin {opacity,mode,..}|SpatialCommand::BeginInSpace {opacity,mode,..}=>
                        opacity==1. && mode==BlendMode::Normal,_=>false,
                };
                if independent && neutral {overwrite=Some(begin);}
                else if frames[depth].2 {frames[depth].3=false;}
            },
            SpatialCommand::RetainedPaint {paint,mode,..}=>{
                if paint.opacity==1. && mode==BlendMode::Normal {overwrite=Some(index);}
                else if frames[depth].2 {frames[depth].3=false;}
            },
            SpatialCommand::Paint(paint)=>{
                if paint.source.alpha()==1. && paint.mode==BlendMode::Normal {overwrite=Some(index);}
                else if frames[depth].2 {frames[depth].3=false;}
            },
            SpatialCommand::SampledPaint {..}|SpatialCommand::NativeSampledPaint {..}=>{
                if frames[depth].2 {frames[depth].3=false;}
            },_=>{},
        }
        if let Some(start)=overwrite {
            for (index,flag) in keep.get_mut(frames[depth].0..start)?.iter_mut().enumerate() {
                if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
                ignored[frames[depth].0+index]=true;
                if *flag {*flag=false;removed+=1;}
            }
            frames[depth].0=start;frames[depth].3=true;
        }
        index+=1;
    }
    if depth!=0 {return None;}Some(removed)
}

/// Remove globally redundant clears before per-coverage route preparation.
pub(super) fn prune_redundant_mask_clears(
    commands:&mut Vec<SpatialCommand<'_>>,remaining:&mut usize,cancelled:&dyn Fn()->bool,
)->Option<usize> {
    let mut saved=[false;33];let mut depth=0usize;let mut active=false;let mut out=0usize;
    for index in 0..commands.len() {
        if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
        let command=commands[index];
        match command {
            SpatialCommand::Begin {..}|SpatialCommand::BeginInSpace {..}=>{
                depth+=1;if depth>=saved.len() {return None;}saved[depth]=active;active=false;
            },
            SpatialCommand::End=>{if depth==0 {return None;}active=saved[depth];depth-=1;},
            SpatialCommand::SetMask {..}|SpatialCommand::SetBudgetedMask {..}=>active=true,
            SpatialCommand::ClearMask=>{if !active {continue;}active=false;},_=>{},
        }
        commands[out]=command;out+=1;
    }
    if depth!=0 {return None;}let removed=commands.len()-out;commands.truncate(out);Some(removed)
}

fn rectangle_strictly_contains_pixel(path:&BezPath,x:i32,y:i32)->bool {
    use kurbo::{PathEl,Point};
    let elements=path.elements();
    if !(5..=6).contains(&elements.len()) {return false;}
    let (PathEl::MoveTo(a),PathEl::LineTo(b),PathEl::LineTo(c),PathEl::LineTo(d))=
        (elements[0],elements[1],elements[2],elements[3]) else {return false;};
    if elements.len()==5 {
        if elements[4]!=PathEl::ClosePath {return false;}
    } else if elements[4]!=PathEl::LineTo(a) || elements[5]!=PathEl::ClosePath {return false;}
    if ![a,b,c,d].iter().all(|p|p.x.is_finite() && p.y.is_finite()) || a.x==c.x || a.y==c.y {return false;}
    if !((b==Point::new(a.x,c.y) && d==Point::new(c.x,a.y))
        || (d==Point::new(a.x,c.y) && b==Point::new(c.x,a.y))) {return false;}
    a.x.min(c.x)<f64::from(x) && a.x.max(c.x)>f64::from(x)+1.
        && a.y.min(c.y)<f64::from(y) && a.y.max(c.y)>f64::from(y)+1.
}

#[test]
fn constant_rectangle_certificate_rejects_boundary_contact_and_nonrectangles() {
    use kurbo::{PathEl,Point};
    assert!(rectangle_strictly_contains_pixel(&kurbo::Rect::new(-1.,-1.,2.,2.).to_path(0.1),0,0));
    assert!(!rectangle_strictly_contains_pixel(&kurbo::Rect::new(0.,0.,1.,1.).to_path(0.1),0,0));
    let diagonal=BezPath::from_vec(vec![PathEl::MoveTo(Point::new(-1.,-1.)),PathEl::LineTo(Point::new(2.,2.)),
        PathEl::LineTo(Point::new(-1.,2.)),PathEl::LineTo(Point::new(2.,-1.)),PathEl::ClosePath]);
    assert!(!rectangle_strictly_contains_pixel(&diagonal,0,0));
    assert!(!rectangle_strictly_contains_pixel(&kurbo::Rect::new(-1.,-1.,2.,2.).to_path(0.1),3,0));
}

fn route_membership_signature(
    commands:&[SpatialCommand<'_>],keep:&[bool],ignored:&[bool],key:&[u64;64],
    remaining:&mut usize,cancelled:&dyn Fn()->bool,
)->Option<[u64;64]> {
    let mut signature=[0u64;64];
    for (index,command) in commands.iter().enumerate() {
        if cancelled() {return None;}*remaining=remaining.checked_sub(1)?;
        if ignored[index] {continue;}
        let required=match command {
            SpatialCommand::Paint(p)=>p.regions,
            SpatialCommand::RetainedPaint {regions,..}|SpatialCommand::SampledPaint {regions,..}|
            SpatialCommand::NativeSampledPaint {regions,..}=>*regions,_=>continue,
        };
        let mut witnessed=false;
        for &region in required {
            *remaining=remaining.checked_sub(1)?;
            if region>=4096 {return None;}
            if keep[index] {signature[region/64]|=1u64<<(region%64);}
            else if key[region/64]&(1u64<<(region%64))==0 {
                signature[region/64]|=1u64<<(region%64);witnessed=true;break;
            }
        }
        if !keep[index] && !witnessed {return None;}
    }
    Some(signature)
}

#[test]
fn route_signature_ignores_hidden_geometry_but_keeps_invisibility_witness() {
    use super::native_blend::Model;
    let source=Pixel::from_native(Model::Rgb,&[0.,1.,0.],1.).unwrap();
    let paint=|regions|SpatialCommand::Paint(SpatialPaint {regions,source,mode:BlendMode::Normal});
    let commands=[paint(&[0]),paint(&[1,2]),paint(&[3])];
    let mut key=[0u64;64];key[0]=(1<<2)|(1<<3);
    let signature=route_membership_signature(&commands,&[false,false,true],&[true,false,false],
        &key,&mut 100,&||false).unwrap();
    assert_eq!(signature[0],(1<<1)|(1<<3));
    // A false paint with no false required region cannot certify invisibility.
    key[0]|=1<<1;
    assert!(route_membership_signature(&commands,&[false,false,true],&[true,false,false],
        &key,&mut 100,&||false).is_none());
}

#[test]
fn opaque_single_paint_group_matches_unoptimized_duplicate_paints() {
    use super::{native_blend::Model,native_paint::NativePaint};
    use hayro_interpret::color::{AlphaColor,Color};
    let admit=|_:usize|Some(Box::new(()) as Box<dyn std::any::Any+Send+Sync>);
    let color=Color::from_rgba(AlphaColor::new([0.2,0.7,0.4,1.]));
    let paint=NativePaint::capture(&color,&||false,&admit).unwrap();
    let path=kurbo::Rect::new(0.,0.,1.,1.).to_path(0.1);
    let regions=[(&path,Fill::NonZero)];
    let backdrop=Pixel::from_native(Model::Rgb,&[0.8,0.3,0.1],0.7).unwrap();
    for parent_knockout in [false,true] {for isolated in [false,true] {
        for child_knockout in [false,true] {for ais in [false,true] {
            for opacity in [0.,0.4,1.] {for mask in [0.,0.3,1.] {
                let sample=|_:kurbo::Point,_:&dyn Fn()->bool|Some(mask);
                let begin=SpatialCommand::Begin {isolated,knockout:child_knockout,
                    opacity,alpha_is_shape:ais,mode:BlendMode::Normal};
                let retained=SpatialCommand::RetainedPaint {regions:&[0],paint:&paint,mode:BlendMode::Normal};
                let mut commands=vec![SpatialCommand::Begin {isolated:false,
                    knockout:parent_knockout,opacity:1.,alpha_is_shape:false,mode:BlendMode::Normal},
                    SpatialCommand::SetMask {sample:&sample,sample_cost:1},begin,retained,
                    SpatialCommand::End,SpatialCommand::ClearMask,SpatialCommand::End];
                let actual=compose_grouped_polygon_pixel(&regions,&commands,&paint.color_space,
                    backdrop,0,0,100000,100000,&||false,&admit).unwrap();
                // Two identical opaque paints are algebraically equivalent,
                // but force the ordinary child-frame execution path.
                commands.insert(4,retained);
                let expected=compose_grouped_polygon_pixel(&regions,&commands,&paint.color_space,
                    backdrop,0,0,100000,100000,&||false,&admit).unwrap();
                assert!((actual.alpha()-expected.alpha()).abs()<1e-12);
                for (a,b) in actual.native_components().unwrap().iter().zip(expected.native_components().unwrap()) {
                    assert!((a-b).abs()<1e-12,"{a} != {b}");
                }
            }}
        }}
    }}
}
